cmake_minimum_required(VERSION 3.20)
project(herta_export_geometry LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The installed geometry ABI targets Windows x64 MSVC")
endif()
add_library(herta_export_geometry_windows_x64 SHARED ../export_geometry.cpp)
target_compile_features(herta_export_geometry_windows_x64 PRIVATE cxx_std_20)
target_compile_options(herta_export_geometry_windows_x64 PRIVATE /O2 /EHsc /fp:precise /utf-8)
set_property(TARGET herta_export_geometry_windows_x64 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
